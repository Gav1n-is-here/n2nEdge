cmake_minimum_required(VERSION 3.16)
project(N2nWindows C)
set(CMAKE_C_STANDARD 99)
set(N2N "${CMAKE_CURRENT_SOURCE_DIR}/vendor/n2n")
file(GLOB CORE "${N2N}/src/*.c")
list(FILTER CORE EXCLUDE REGEX "/(sn_management|sn_utils|edge|supernode|example[^/]*|benchmark)\\.c$")
add_executable(edge ${CORE} entry.c "${N2N}/win32/wintap.c")
target_include_directories(edge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/compat" "${N2N}/include" "${N2N}/win32" "${N2N}")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
 set(EDGE_OS "Windows-x64")
else()
 set(EDGE_OS "Windows-x86")
endif()
target_compile_definitions(edge PRIVATE WIN32 CMAKE_BUILD N2N_VERSION="3.0.0" N2N_OSNAME="${EDGE_OS}" SKIP_MULTICAST_PEERS_DISCOVERY)
target_compile_options(edge PRIVATE -Wno-incompatible-pointer-types -Wno-implicit-function-declaration)
target_link_options(edge PRIVATE -static)
target_link_libraries(edge PRIVATE ws2_32 iphlpapi advapi32)
